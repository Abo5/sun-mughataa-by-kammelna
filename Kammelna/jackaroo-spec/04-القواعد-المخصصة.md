## ٣. ⭐ خريطة التطبيق — `IsCustomRule` @`0x51DB948` ✅

```csharp
if (allowedRules == 0) { actions = NONE; return false; }   // كل شيء افتراضي

switch (card.Rank)
{
 case JACK (9):
   if (IsRed(card))   { actions = Has(JACK_RED_SWAP /*0x1000*/) ? SWAP : MOVE;
                        return Has(JACK_RED_SWAP); }
   if (IsBlack(card)) { actions = Has(JACK_BLACK_MOVE /*0x800*/) ? MOVE : SWAP;
                        return Has(JACK_BLACK_MOVE); }

 case QUEEN (10):
   if (IsRed(card)) {
       if (Has(QUEEN_RED_MOVE /*0x200*/))  { actions = MOVE; return true; }
       actions = Has(QUEEN_RED_STEAL /*0x400*/) ? STEAL : BURN_OTHER_CARD_RANDOM;
       return Has(QUEEN_RED_STEAL);
   }
   if (IsBlack(card)) {
       if (Has(QUEEN_BLACK_BURN /*0x80*/)) { actions = BURN_OTHER_CARD_RANDOM; return true; }
       actions = Has(QUEEN_BLACK_STEAL /*0x100*/) ? STEAL : MOVE;
       return Has(QUEEN_BLACK_STEAL);
   }

 case KING (11):
   if (IsRed(card)) {
       if (Has(KING_RED_PLANT /*0x40*/))         { actions = PLANT;         return true; }
       if (Has(KING_RED_MOVE_EAT_PATH /*0x20*/)) { actions = MOVE_EAT_PATH; return true; }
       actions = Has(KING_RED_PLANT_MOVE_NO_KILL /*0x10*/)
               ? (PLANT | MOVE)              // 0x20001 — ١٣ خطوة بلا أكل الطريق
               : (PLANT | MOVE_EAT_PATH);    // 0x20008
       return Has(KING_RED_PLANT_MOVE_NO_KILL);
   }
   if (IsBlack(card)) {   // نفس البنية بأعلام 8 / 4 / 2
       if (Has(KING_BLACK_PLANT /*8*/))         { actions = PLANT;         return true; }
       if (Has(KING_BLACK_MOVE_EAT_PATH /*4*/)) { actions = MOVE_EAT_PATH; return true; }
       actions = Has(KING_BLACK_PLANT_MOVE_NO_KILL /*2*/) ? (PLANT|MOVE) : (PLANT|MOVE_EAT_PATH);
       return Has(KING_BLACK_PLANT_MOVE_NO_KILL);
   }

 case TEN (8):
   actions = Has(TEN_MOVE /*0x2000*/) ? MOVE : (MOVE | BURN_OTHER_CARD);   // 0x8001
   return Has(TEN_MOVE);

 case ACE (12):
   actions = Has(ACE_MOVE_1_PLANT /*1*/)
           ? (PLANT | MOVE_CHOICE_1)                       // 0x20010 — ١ فقط
           : (PLANT | MOVE_CHOICE_1 | MOVE_CHOICE_11);     // 0x20030 — ١ أو ١١
   return Has(ACE_MOVE_1_PLANT);

 default: actions = NONE; return false;      // بقية الأوراق لا تتأثر بـ jcr
}
```

> **الأوراق المتأثّرة بـ`jcr`: ١٠ · ولد · بنت · شايب · إس فقط.**
> والأعلام `0x4000` · `0x8000` · `0x10000` · `0x20000` · `0x40000` تؤثّر على **الحجب**
> لا على الورق — انظر [٠٢ §٦](02-الحركة-والأكل.md).
